The core of the code

      
Age sets growth, weight sets thickness, tasks set pose targets, and cosmetics attach to anchors.


// weight index: -1 underweight .. 0 healthy .. +1 overweight
w = clamp(w - DECAY_PER_DAY * days + MEAL_GAIN * meals, -1, 1);
const m = Math.abs(w) ** 1.5;                 // subtle near 0, obvious at the ends
const thick = w >= 0 ? 1 + 0.5 * m : 1 - 0.3 * m;
bodyH = breed.bodyH * 50 * thick;

// idle: pick the next neutral state at random
const next = weightedPick(['still', 'pant|purr', 'nap', 'walk'], except(current));

// every action is a set of pose targets, eased so changes never pop
const T = { sit: 0, lie: 0, paw: 0, headDown: 0, eyes: 0, squint: 0, pant: 0 };
pose[k] += (T[k] - pose[k]) * (1 - Math.exp(-dt * rate));

// cosmetics are drawn from anchors the pet exposes each frame
anchors = { head: { x, y, pitch, r }, neck: { x, y }, torso: { theta, dy },
            paws: { foreNear, foreFar, hindNear, hindFar } };
drawHat(ctx, anchors.head, wear.hat);         // follows sit, sleep, bark, growth

// scenes are data: new rooms need no code
scene = { sky, layers, floor: { horizon, yMin, yMax }, props: [{ kind, x, y }],
          pois: [{ type: 'food' | 'water' | 'bed' | 'home', x, y }], ambient };
const bowl = findPoi('food');                 // actions look up spots by type, not by scene
