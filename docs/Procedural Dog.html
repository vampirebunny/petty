The core of the code

      
Each action sets pose targets, the pose eases toward them, and legs are solved from where the paws should be.


// every behaviour is a set of pose targets
const T = { sit: 0, lie: 0, paw: 0, headDown: 0, eyes: 0, pant: 0 };
if (task === 'sit')   T.sit = 1;
if (task === 'sleep') { T.lie = 1; T.eyes = 1; }
if (task === 'paw')   { T.sit = 1; T.paw = 1; }

// ease the pose toward the targets so changes never pop
for (const k in T) pose[k] += (T[k] - pose[k]) * (1 - Math.exp(-dt * 6));

// two-bone IK: say where the paw goes, solve for the knee
function ik(hip, foot, upper, lower, side) {
  const d = Math.min(Math.hypot(foot.x - hip.x, foot.y - hip.y), upper + lower - 0.01);
  const a = Math.acos((upper * upper + d * d - lower * lower) / (2 * upper * d));
  const dir = Math.atan2(foot.y - hip.y, foot.x - hip.x) + side * a;
  return { x: hip.x + Math.cos(dir) * upper, y: hip.y + Math.sin(dir) * upper };
}
